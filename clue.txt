#Инициилизация терраформа
terraform init
#Обновление версии и связанных хешей
terraform init -upgrade
#Сброс Прокси для того что бы терминал смог работать с локальными ВМ и контейнерами
$env:HTTP_PROXY="" 
$env:HTTPS_PROXY=""
#Валидация кода перед запуском терраформа
terraform validate
#Планирование изменений проверка перед запуском через создание плана
terraform plan -out [имя_плана].tfplan
#Планирование изменений проверка перед запуском
terraform plan
#Запуск скрипта для изменения/создания ВМ и контейнеров по созданному плану
terraform apply plan_2026.07.06.tfplan
#Запуск скрипта для изменения/создания ВМ и контейнеров 
terraform apply
#Запуск скрипта для изменения/создания ВМ и контейнеров по созданному плану с ограничением на активность по 5 машин в моменте
terraform apply plan_2026.07.06.tfplan -parallelism=5
#Запуск скрипта для изменения/создания ВМ и контейнеров с ограничением на активность по 5 машин в моменте
terraform apply -parallelism=5
#Запуск скрипта для удаление ВМ и контейнеров
terraform destroy
#Запуск скрипта для удаления ВМ и контейнеров с ограничением на активность по 5 машин в моменте
terraform destroy -parallelism=5
